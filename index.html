🚀 FAST & PURE CLIENT-SIDE

Coursera Fast Certificate Extractor

Bypasses API Cloudflare & HTML login verification challenges. Extracts every completed course title and certificate directly from the loaded DOM in seconds.

💡 Why Direct DOM Extraction?

Calling Coursera's internal REST API endpoints via `fetch()` often throws: 

SyntaxError: Unexpected token '<', "<html>..." is not valid JSON

Coursera protects internal API endpoints behind HTML verification, security challenges, and session routing. However, when you browse your Completed tab, the browser has already authenticated and rendered every course into the DOM. Our script extracts directly from the page—zero API calls, zero login tokens required, 100% private.

Lightweight, robust DOM extractor. Prints a table and triggers a CSV download: 

(() => {
  const records = [];
  const seen = new Set();

  document.querySelectorAll('a').forEach(a => {
    const href = a.href || '';
    const text = (a.innerText || '').trim();

    const isMatch = href.includes('/accomplishments') || 
                    href.includes('/certificate') || 
                    href.includes('/learn/') ||
                    text.toLowerCase().includes('view certificate') ||
                    text.toLowerCase().includes('completed');

    if (isMatch) {
      let container = a.parentElement;
      for (let i = 0; i < 6 && container; i++) {
        const heading = container.querySelector('h2, h3, h4, [data-test*="title"], [data-testid*="title"]');
        if (heading && heading.innerText.trim()) {
          const title = heading.innerText.trim().split('\n')[0];
          if (!seen.has(title) && title.length > 3 && !['My Learning', 'Completed', 'In Progress', 'Certificates'].includes(title)) {
            seen.add(title);
            records.push({
              Title: title,
              URL: href.includes('http') ? href : window.location.origin + href
            });
          }
          break;
        }
        container = container.parentElement;
      }
    }
  });

  if (records.length === 0) {
    document.querySelectorAll('h3, h2, h4').forEach(h => {
      const text = h.innerText.trim();
      const forbidden = ['My Learning', 'Completed', 'In Progress', 'Certificates', 'Browse', 'Explore'];
      if (text.length > 4 && !forbidden.includes(text) && !seen.has(text)) {
        seen.add(text);
        const parentLink = h.closest('a') || (h.parentElement ? h.parentElement.querySelector('a') : null);
        records.push({
          Title: text,
          URL: parentLink ? parentLink.href : 'Completed Tab'
        });
      }
    });
  }

  if (records.length === 0) {
    console.warn("No courses detected. Make sure you scrolled down so cards render in the DOM.");
    return;
  }

  console.table(records);

  const csvContent = "data:text/csv;charset=utf-8," 
    + ["Title,URL"].concat(records.map(r => `"${r.Title.replace(/"/g, '""')}","${r.URL}"`)).join("\n");
  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const downloadUrl = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = downloadUrl;
  a.download = "coursera_completed_summary.csv";
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);

  console.log(`%c Successfully captured ${records.length} items!`, "color: #10b981; font-weight: bold; font-size: 14px;");
})();

Extracts Provider, Grade, Credential ID, and injects an on-screen modal banner on Coursera with JSON & Markdown buttons: 

(() => {
  const records = [];
  const seen = new Set();
  console.log("%c🔍 Scanning Coursera Completed tab DOM...", "color: #3b82f6; font-weight: bold; font-size: 14px;");

  const cleanText = (str) => (str || '').replace(/\s+/g, ' ').trim();
  const extractCredentialId = (url) => {
    if (!url) return '';
    const match = url.match(/(?:verify|accomplishments\/(?:specialization|certificate)?)\/([a-zA-Z0-9_-]+)/);
    return match ? match[1] : '';
  };
  const extractDate = (text) => {
    const match = text.match(/(?:Completed|Earned|Finished|Issued)?\s*([A-Za-z]+ \d{1,2}, \d{4}|\d{1,2}\/\d{1,2}\/\d{4})/i);
    return match ? match[1] : '';
  };
  const extractGrade = (text) => {
    const match = text.match(/(\d{1,3}(?:\.\d{1,2})?%)/);
    return match ? match[1] : '';
  };

  const anchors = Array.from(document.querySelectorAll('a'));
  anchors.forEach(a => {
    const href = a.href || '';
    const text = cleanText(a.innerText);
    const isCert = href.includes('/accomplishments') || href.includes('/certificate') || href.includes('/account/accomplishments') || text.toLowerCase().includes('view certificate');

    if (isCert) {
      let container = a.parentElement;
      for (let i = 0; i < 7 && container; i++) {
        const titleEl = container.querySelector('h2, h3, h4, [data-test*="title"], [data-testid*="title"]');
        if (titleEl && cleanText(titleEl.innerText)) {
          const title = cleanText(titleEl.innerText).split('\n')[0];
          const fullText = cleanText(container.innerText);
          const forbidden = ['my learning', 'completed', 'in progress', 'certificates', 'browse', 'explore'];

          if (!seen.has(title.toLowerCase()) && title.length > 3 && !forbidden.includes(title.toLowerCase())) {
            seen.add(title.toLowerCase());
            let provider = '';
            const partnerEl = container.querySelector('[data-test*="partner"], [data-testid*="partner"], .partner-name');
            if (partnerEl) {
              provider = cleanText(partnerEl.innerText);
            }
            const certUrl = href.startsWith('http') ? href : window.location.origin + href;
            records.push({
              title,
              provider: provider || 'Coursera Partner',
              grade: extractGrade(fullText) || 'N/A',
              date: extractDate(fullText) || 'N/A',
              credentialId: extractCredentialId(certUrl) || 'N/A',
              url: certUrl
            });
          }
          break;
        }
        container = container.parentElement;
      }
    }
  });

  if (records.length === 0) {
    alert("No completed courses found! Please scroll down to load cards first.");
    return;
  }

  console.table(records);

  const csvHeaders = ["Title", "Provider", "Grade", "Completion Date", "Credential ID", "Certificate URL"];
  const csvRows = records.map(r => [
    `"${r.title.replace(/"/g, '""')}"`,
    `"${r.provider.replace(/"/g, '""')}"`,
    `"${r.grade.replace(/"/g, '""')}"`,
    `"${r.date.replace(/"/g, '""')}"`,
    `"${r.credentialId.replace(/"/g, '""')}"`,
    `"${r.url.replace(/"/g, '""')}"`
  ].join(","));
  const csvData = "\uFEFF" + [csvHeaders.join(",")].concat(csvRows).join("\r\n");

  const blob = new Blob([csvData], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = "coursera_completed_courses.csv";
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);

  alert(`🎉 Captured ${records.length} courses! CSV downloaded.`);
})();

Run this first if you have 50+ courses to smoothly scroll and render all lazy cards: 

(() => {
  console.log("%c Starting Coursera Auto-Scroll...", "color: #3b82f6; font-weight: bold;");
  let lastScrollHeight = 0, noChangeCount = 0;
  const timer = setInterval(() => {
    const currentHeight = document.documentElement.scrollHeight;
    window.scrollTo({ top: currentHeight, behavior: 'smooth' });
    if (currentHeight === lastScrollHeight) {
      if (++noChangeCount >= 4) {
        clearInterval(timer);
        window.scrollTo({ top: 0, behavior: 'smooth' });
        console.log("%c Auto-scroll finished! Now run the extractor script.", "color: #10b981; font-weight: bold; font-size: 14px;");
      }
    } else {
      noChangeCount = 0;
      lastScrollHeight = currentHeight;
    }
  }, 800);
})();

🖨️ Zero-Code Fallback Method

If corporate restrictions, ad-blockers, or browser policies prevent running scripts in DevTools:

  1. Navigate to the Completed tab and scroll down until all your cards appear.
  2. Press Ctrl + P (or Cmd + P on macOS).
  3. Select Save as PDF as the destination printer.
  4. Save the document. You now have a complete, permanent, searchable archive of all your certificates!
