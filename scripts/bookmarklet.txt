# Coursera Extractor Bookmarklet (1-Click)

You can save this code as a browser bookmark. When you are on your Coursera Completed tab, clicking the bookmark will immediately extract all courses and trigger the CSV download!

### How to Install:
1. Open your browser's Bookmarks Manager (`Ctrl + Shift + O` on Windows/Linux or `Cmd + Option + B` on Mac).
2. Right-click and choose **Add new bookmark**.
3. Name it: `🎓 Coursera Extractor`
4. Paste the code below into the **URL** field:

```javascript
javascript:(function(){const records=[];const seen=new Set();document.querySelectorAll('a').forEach(a=>{const href=a.href||'';const text=(a.innerText||'').trim();if(href.includes('/accomplishments')||href.includes('/certificate')||href.includes('/learn/')||text.toLowerCase().includes('view certificate')||text.toLowerCase().includes('completed')){let container=a.parentElement;for(let i=0;i<6&&container;i++){const heading=container.querySelector('h2,h3,h4,[data-test*="title"],[data-testid*="title"]');if(heading&&heading.innerText.trim()){const title=heading.innerText.trim().split('\n')[0];if(!seen.has(title)&&title.length>3&&!['My Learning','Completed','In Progress','Certificates'].includes(title)){seen.add(title);records.push({Title:title,URL:href.includes('http')?href:window.location.origin+href});}break;}container=container.parentElement;}}});if(records.length===0){document.querySelectorAll('h3,h2,h4').forEach(h=>{const text=h.innerText.trim();if(text.length>4&&!['My Learning','Completed','In Progress','Certificates','Browse','Explore'].includes(text)&&!seen.has(text)){seen.add(text);const parentLink=h.closest('a')||(h.parentElement?h.parentElement.querySelector('a'):null);records.push({Title:text,URL:parentLink?parentLink.href:'Completed Tab'});}});}if(records.length===0){alert('No courses found! Please scroll down to load your completed course cards first.');return;}const csvContent='data:text/csv;charset=utf-8,'+['Title,URL'].concat(records.map(r=>`"${r.Title.replace(/"/g,'""')}","${r.URL}"`)).join('\n');const blob=new Blob([csvContent],{type:'text/csv;charset=utf-8;'});const downloadUrl=URL.createObjectURL(blob);const a=document.createElement('a');a.href=downloadUrl;a.download='coursera_completed_summary.csv';document.body.appendChild(a);a.click();document.body.removeChild(a);alert(`🎉 Successfully extracted ${records.length} Coursera courses!\nYour CSV file has downloaded.`);})();
```
